    <script>
        function formatDate(d) {
            return `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')}`;
        }

        function getMonday(d) {
            const date = new Date(d);
            const day = date.getDay();
            const diff = date.getDate() - (day === 0 ? 6 : day - 1);
            return new Date(date.setDate(diff));
        }

        const todayObj = new Date();
        const todayStr = formatDate(todayObj);

        let state = {
            profileName: localStorage.getItem('profileName') || "Barbería de Sergio",
            selectedDate: todayStr,
            calendarWeekStart: getMonday(todayObj),
            appointments: JSON.parse(localStorage.getItem('appointments')) || [],
            clients: JSON.parse(localStorage.getItem('clients')) || []
        };

        function saveData() {
            localStorage.setItem('profileName', state.profileName);
            localStorage.setItem('appointments', JSON.stringify(state.appointments));
            localStorage.setItem('clients', JSON.stringify(state.clients));
        }

        document.addEventListener('DOMContentLoaded', () => {
            updateProfileUI();
            renderWeeklyCalendar();
            renderAppointments();
            renderClients();
        });

        function updateProfileUI() {
            const name = state.profileName;
            document.getElementById('profileNameDisplay').textContent = name;
            document.getElementById('profileNameInput').value = name;
            document.getElementById('userAvatar').textContent = name.charAt(0).toUpperCase();
        }

        function saveProfileSettings() {
            const newName = document.getElementById('profileNameInput').value.trim();
            if (newName) {
                state.profileName = newName;
                saveData();
                updateProfileUI();
                alert('¡Nombre del perfil guardado correctamente!');
            }
        }

        function renderWeeklyCalendar() {
            const grid = document.getElementById('weeklyCalendarGrid');
            grid.innerHTML = '';
            const dayNames = ['Lu', 'Ma', 'Mi', 'Ju', 'Vi', 'Sá', 'Do'];
            const startMonday = new Date(state.calendarWeekStart);
            const monthNames = ["Enero", "Febrero", "Marzo", "Abril", "Mayo", "Junio", "Julio", "Agosto", "Septiembre", "Octubre", "Noviembre", "Diciembre"];
            
            document.getElementById('calendarMonthTitle').textContent = `${monthNames[startMonday.getMonth()]} ${startMonday.getFullYear()}`;

            for (let i = 0; i < 7; i++) {
                const currentDay = new Date(startMonday);
                currentDay.setDate(startMonday.getDate() + i);
                const dayStr = formatDate(currentDay);
                const isSelected = dayStr === state.selectedDate;

                const dayCol = document.createElement('div');
                dayCol.className = 'flex flex-col items-center space-y-2 cursor-pointer';
                dayCol.onclick = () => selectDate(dayStr);
                dayCol.innerHTML = `
                    <span class="text-xs text-gray-400 font-medium">${dayNames[i]}</span>
                    <div class="w-10 h-10 flex items-center justify-center text-sm font-bold rounded-xl transition ${isSelected ? 'bg-[#587b6a] text-white shadow-sm' : 'text-gray-800 hover:bg-gray-100'}">
                        ${currentDay.getDate()}
                    </div>
                `;
                grid.appendChild(dayCol);
            }
        }

        function selectDate(dateStr) {
            state.selectedDate = dateStr;
            renderWeeklyCalendar();
            renderAppointments();
        }

        function selectQuickDate(type) {
            const realToday = new Date();
            if (type === 'tomorrow') realToday.setDate(realToday.getDate() + 1);
            state.selectedDate = formatDate(realToday);
            state.calendarWeekStart = getMonday(realToday);
            renderWeeklyCalendar();
            renderAppointments();
        }

        function navigateWeek(direction) {
            state.calendarWeekStart.setDate(state.calendarWeekStart.getDate() + (direction * 7));
            renderWeeklyCalendar();
        }

        function renderAppointments() {
            document.getElementById('agendaHeaderTitle').textContent = `Agenda del ${state.selectedDate}`;
            const filtered = state.appointments.filter(a => a.date === state.selectedDate);
            const container = document.getElementById('appointmentsContainer');
            document.getElementById('appointmentSummary').textContent = `${filtered.length} citas agendadas para esta fecha`;

            const dailyIncome = filtered.filter(a => a.status === 'Completada').reduce((sum, a) => sum + a.price, 0);
            document.getElementById('dailyIncomeDisplay').textContent = `€${dailyIncome} EUR`;

            if (filtered.length === 0) {
                container.innerHTML = `
                    <div class="border-2 border-dashed border-gray-100 rounded-2xl p-10 text-center space-y-4 my-2">
                        <div class="w-12 h-12 rounded-2xl bg-gray-50 flex items-center justify-center mx-auto text-gray-400 text-xl">📅</div>
                        <p class="font-medium text-gray-700 text-sm">¡Todo listo! No hay citas programadas para este día.</p>
                        <button type="button" onclick="openModal()" class="bg-[#587b6a] hover:bg-[#466355] text-white font-semibold text-xs px-5 py-2.5 rounded-xl transition shadow-sm inline-flex items-center gap-1.5 cursor-pointer">
                            + Agendar cita
                        </button>
                    </div>
                `;
                return;
            }

            let html = `<div class="space-y-3">`;
            filtered.forEach(app => {
                const isCompleted = app.status === 'Completada';
                html += `
                    <div class="flex flex-wrap items-center justify-between p-4 bg-gray-50 rounded-xl border border-gray-100 gap-4">
                        <div class="flex items-center gap-4">
                            <span class="font-bold text-gray-900 bg-white px-3 py-1.5 rounded-lg border text-sm">${app.time}</span>
                            <div>
                                <h4 class="font-bold text-gray-900 text-sm">${app.clientName}</h4>
                                <p class="text-xs text-gray-500">${app.serviceName} • €${app.price} EUR</p>
                            </div>
                        </div>
                        <div class="flex items-center gap-2">
                            <span class="px-2.5 py-1 text-xs font-bold rounded-full ${isCompleted ? 'bg-emerald-100 text-emerald-800' : 'bg-amber-100 text-amber-800'}">${app.status}</span>
                            ${!isCompleted ? `<button onclick="toggleAppointmentStatus(${app.id}, 'Completada')" class="text-xs bg-emerald-600 text-white px-3 py-1.5 rounded-lg font-semibold">Marcar Completada</button>` : ''}
                            <button onclick="deleteAppointment(${app.id})" class="text-xs text-red-500 px-2 py-1">Eliminar</button>
                        </div>
                    </div>
                `;
            });
            html += `</div>`;
            container.innerHTML = html;
        }

        function openModal() { document.getElementById('appointmentModal').classList.remove('hidden'); }
        function closeModal() { document.getElementById('appointmentModal').classList.add('hidden'); }

        function handleCreateAppointment(e) {
            e.preventDefault();
            const clientName = document.getElementById('clientName').value.trim();
            const clientContact = document.getElementById('clientContact').value.trim();
            const serviceName = document.getElementById('serviceName').value.trim();
            const time = document.getElementById('appointmentTime').value;
            const price = parseFloat(document.getElementById('appointmentPrice').value) || 0;

            const newAppointment = { id: Date.now(), date: state.selectedDate, clientName, clientContact, serviceName, time, price, status: 'Pendiente' };
            state.appointments.push(newAppointment);

            let client = state.clients.find(c => c.name.toLowerCase() === clientName.toLowerCase());
            if (!client) { state.clients.push({ name: clientName, contact: clientContact, totalCount: 1 }); } 
            else { client.totalCount += 1; }

            saveData();
            closeModal();
            renderAppointments();
            renderClients();
        }

        function toggleAppointmentStatus(id, status) {
            const app = state.appointments.find(a => a.id === id);
            if (app) { app.status = status; saveData(); renderAppointments(); }
        }

        function deleteAppointment(id) {
            if (confirm('¿Seguro que deseas eliminar esta cita?')) {
                state.appointments = state.appointments.filter(a => a.id !== id);
                saveData();
                renderAppointments();
            }
        }

        function renderClients() {
            document.getElementById('clientCountNav').textContent = state.clients.length;
            const tbody = document.getElementById('clientsTableBody');
            if (state.clients.length === 0) {
                tbody.innerHTML = `<tr><td colspan="3" class="py-4 text-center text-xs text-gray-400">No hay clientes registrados aún.</td></tr>`;
                return;
            }
            tbody.innerHTML = state.clients.map(c => `
                <tr class="hover:bg-gray-50 transition">
                    <td class="py-3 px-2 font-medium text-gray-900">${c.name}</td>
                    <td class="py-3 px-2 text-gray-500">${c.contact}</td>
                    <td class="py-3 px-2 text-gray-700 font-semibold">${c.totalCount}</td>
                </tr>
            `).join('');
        }

        function switchTab(tabId) {
            document.querySelectorAll('.view-panel').forEach(p => p.classList.add('hidden'));
            document.querySelectorAll('.tab-btn').forEach(b => {
                b.classList.remove('bg-white', 'text-gray-900', 'shadow-sm');
                b.classList.add('text-gray-600');
            });
            document.getElementById(`view-${tabId}`).classList.remove('hidden');
            const activeBtn = document.getElementById(`tab-${tabId}`);
            activeBtn.classList.add('bg-white', 'text-gray-900', 'shadow-sm');
            activeBtn.classList.remove('text-gray-600');
        }

        function setLang(lang) {
            const esBtn = document.getElementById('lang-es');
            const enBtn = document.getElementById('lang-en');
            if (lang === 'ES') {
                esBtn.className = "px-2.5 py-1 rounded-full bg-black text-white";
                enBtn.className = "px-2.5 py-1 rounded-full text-gray-500 hover:text-black";
            } else {
                enBtn.className = "px-2.5 py-1 rounded-full bg-black text-white";
                esBtn.className = "px-2.5 py-1 rounded-full text-gray-500 hover:text-black";
            }
        }

        function logout() {
            if (confirm('¿Deseas cerrar la sesión?')) alert('Sesión cerrada correctamente.');
        }
    </script>
